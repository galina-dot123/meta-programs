export default function App() {

  const [step, setStep] = useState(0); // 0 = Intro, 1-5 = Domains, 6 = Results

  const [answers, setAnswers] = useState({});

  const [errorMsg, setErrorMsg] = useState('');


  const handleAnswer = (qId, value) => {

    setAnswers(prev => ({ ...prev, [qId]: value }));

    setErrorMsg('');

  };


  const handleNext = () => {

    if (step >= 1 && step <= 5) {

      const currentDomain = domains[step - 1];

      const unanswered = currentDomain.questions.filter(q => !answers[q.id]);

      if (unanswered.length > 0) {

        setErrorMsg('Пожалуйста, ответьте на все вопросы на этой странице, чтобы продолжить.');

        return;

      }

    }

    window.scrollTo(0, 0);

    setStep(prev => prev + 1);

  };


  const calculateResults = useMemo(() => {

    if (step !== 6) return null;


    const results = {

      domains: [],

      metaprograms: []

    };


    let allMps = [];


    domains.forEach(domain => {

      let domainTotalScore = 0;

      let domainMpCount = domain.questions.length / 2;

      

      // Сгруппировать вопросы по метапрограммам (по id числа)

      const mpMap = {};

      domain.questions.forEach(q => {

        if (!mpMap[q.mp]) {

          mpMap[q.mp] = { id: q.mp, name: q.mpName, domainId: domain.id, domainName: domain.title, need: domain.need };

        }

        if (q.type === 'direct') {

          mpMap[q.mp].directScore = answers[q.id] || 0;

        } else {

          // Инверсия для обратных

          const rawScore = answers[q.id] || 0;

          mpMap[q.mp].reverseScore = rawScore > 0 ? (6 - rawScore) : 0;

        }

      });


      Object.values(mpMap).forEach(mp => {

        mp.total = mp.directScore + mp.reverseScore;

        domainTotalScore += mp.total;

        

        let severity = 'Низкая (не проявлена)';

        let color = 'text-green-600';

        let bg = 'bg-green-100';

        if (mp.total >= 8) {

          severity = 'Ярко выражена';

          color = 'text-red-600';

          bg = 'bg-red-100';

        } else if (mp.total >= 5) {

          severity = 'Умеренная';

          color = 'text-yellow-600';

          bg = 'bg-yellow-100';

        }


        mp.severity = severity;

        mp.color = color;

        mp.bg = bg;

        allMps.push(mp);

      });


      results.domains.push({

        id: domain.id,

        title: domain.title,

        need: domain.need,

        averageScore: (domainTotalScore / domainMpCount).toFixed(1)

      });

    });


    results.metaprograms = allMps.sort((a, b) => b.total - a.total); // Сортировка по убыванию выраженности

    

    // Сортировка доменов по убыванию среднего балла

    results.domains.sort((a, b) => parseFloat(b.averageScore) - parseFloat(a.averageScore));


    return results;

  }, [step, answers]);


  const resetTest = () => {

    setAnswers({});

    setStep(0);

    window.scrollTo(0, 0);

  };


  return (

    <div className="min-h-screen bg-gray-50 text-gray-800 font-sans pb-12">

      <header className="bg-indigo-600 text-white p-6 shadow-md">

        <div className="max-w-4xl mx-auto">

          <h1 className="text-2xl font-bold">Архитектура Метапрограмм</h1>

          <p className="text-indigo-200 mt-1">Психометрический профиль личности (33 паттерна)</p>

        </div>

      </header>


      <main className="max-w-4xl mx-auto mt-8 p-4">

        {step === 0 && (

          <div className="bg-white p-8 rounded-2xl shadow-sm border border-gray-100">

            <h2 className="text-3xl font-semibold mb-4 text-gray-900">Инструкция</h2>

            <div className="prose prose-indigo max-w-none text-gray-600 space-y-4 text-lg">

              <p>Вам будет предложено 66 утверждений. Прочитайте каждое и оцените, насколько оно описывает ваши привычные мысли, чувства или поведение.</p>

              <p>Отвечайте <strong>быстро</strong>, опираясь на первое впечатление.</p>

              <div className="bg-indigo-50 p-6 rounded-xl border border-indigo-100 mt-6">

                <h3 className="font-semibold text-indigo-900 mb-2">Шкала оценки:</h3>

                <ul className="list-disc pl-5 space-y-2 text-indigo-800">

                  <li><strong>1</strong> — Абсолютно не согласен</li>

                  <li><strong>2</strong> — Скорее не согласен</li>

                  <li><strong>3</strong> — Нейтрально (иногда да, иногда нет)</li>

                  <li><strong>4</strong> — Скорее согласен</li>

                  <li><strong>5</strong> — Полностью согласен</li>

                </ul>

              </div>

            </div>

            <button 

              onClick={handleNext}

              className="mt-8 w-full md:w-auto px-8 py-4 bg-indigo-600 hover:bg-indigo-700 text-white font-bold rounded-xl transition-all shadow-md hover:shadow-lg transform hover:-translate-y-0.5"

            >

              Начать тестирование

            </button>

          </div>

        )}


        {step >= 1 && step <= 5 && (

          <div className="animate-fade-in">

            <div className="mb-6 flex items-center justify-between text-sm font-medium text-gray-500">

              <span>Шаг {step} из 5</span>

              <span>Осталось: {5 - step}</span>

            </div>

            <div className="w-full bg-gray-200 rounded-full h-2.5 mb-8">

              <div className="bg-indigo-600 h-2.5 rounded-full transition-all duration-500" style={{ width: `${(step / 5) * 100}%` }}></div>

            </div>


            <div className="bg-white p-6 md:p-10 rounded-2xl shadow-sm border border-gray-100">

              <h2 className="text-2xl font-bold text-gray-900 mb-2">{domains[step - 1].title}</h2>

              <p className="text-indigo-600 font-medium mb-8">Исследуемая потребность: {domains[step - 1].need}</p>


              <div className="space-y-12">

                {domains[step - 1].questions.map((q, index) => (

                  <div key={q.id} className="border-b border-gray-100 pb-8 last:border-0 last:pb-0">

                    <p className="text-lg font-medium text-gray-800 mb-6">{index + 1}. {q.text}</p>

                    

                    <div className="flex flex-col sm:flex-row justify-between items-center gap-4 w-full max-w-2xl mx-auto">

                      <span className="text-xs text-gray-400 font-medium w-full sm:w-auto text-left sm:text-right">Абсолютно<br className="hidden sm:block"/>не согласен</span>

                      

                      <div className="flex justify-between items-center w-full sm:w-auto gap-2 md:gap-4">

                        {[1, 2, 3, 4, 5].map(val => (

                          <button

                            key={val}

                            onClick={() => handleAnswer(q.id, val)}

                            className={`w-12 h-12 md:w-14 md:h-14 rounded-full flex items-center justify-center text-lg font-semibold transition-all duration-200 

                              ${answers[q.id] === val 

                                ? 'bg-indigo-600 text-white shadow-md transform scale-110' 

                                : 'bg-gray-100 text-gray-600 hover:bg-indigo-100 hover:text-indigo-700'}`}

                          >

                            {val}

                          </button>

                        ))}

                      </div>


                      <span className="text-xs text-gray-400 font-medium w-full sm:w-auto text-right sm:text-left">Полностью<br className="hidden sm:block"/>согласен</span>

                    </div>

                  </div>

                ))}

              </div>


              {errorMsg && (

                <div className="mt-8 p-4 bg-red-50 text-red-700 rounded-xl border border-red-200 flex items-center">

                  <svg className="w-5 h-5 mr-2 flex-shrink-0" fill="currentColor" viewBox="0 0 20 20"><path fillRule="evenodd" d="M10 18a8 8 0 100-16 8 8 0 000 16zM8.707 7.293a1 1 0 00-1.414 1.414L8.586 10l-1.293 1.293a1 1 0 101.414 1.414L10 11.414l1.293 1.293a1 1 0 001.414-1.414L11.414 10l1.293-1.293a1 1 0 00-1.414-1.414L10 8.586 8.707 7.293z" clipRule="evenodd" /></svg>

                  {errorMsg}

                </div>

              )}


              <div className="mt-10 flex justify-end">

                <button 

                  onClick={handleNext}

                  className="px-8 py-4 bg-indigo-600 hover:bg-indigo-700 text-white font-bold rounded-xl transition-all shadow-md"

                >

                  {step === 5 ? 'Завершить и показать результаты' : 'Следующий домен'}

                </button>

              </div>

            </div>

          </div>

        )}


        {step === 6 && calculateResults && (

          <div className="animate-fade-in space-y-8">

            <div className="bg-white p-8 rounded-2xl shadow-sm border border-gray-100 text-center">

              <h2 className="text-3xl font-bold text-gray-900 mb-2">Ваш Психометрический Профиль</h2>

              <p className="text-gray-500">Анализ завершен. Ознакомьтесь с выявленными паттернами.</p>

            </div>


            {/* Блок доменов */}

            <div className="bg-white p-8 rounded-2xl shadow-sm border border-gray-100">

              <h3 className="text-xl font-bold text-gray-800 mb-6 border-b pb-4">Ключевая неудовлетворенная потребность</h3>

              

              <div className="space-y-6">

                {calculateResults.domains.map((domain, index) => (

                  <div key={domain.id} className={`p-5 rounded-xl border ${index === 0 ? 'bg-indigo-50 border-indigo-200' : 'bg-gray-50 border-gray-100'}`}>

                    <div className="flex justify-between items-start mb-2">

                      <div>

                        <h4 className={`font-bold ${index === 0 ? 'text-indigo-900 text-lg' : 'text-gray-800'}`}>

                          {index === 0 && '🥇 '} {domain.title.replace('ДОМЕН ', '')}

                        </h4>

                        <p className={`text-sm mt-1 ${index === 0 ? 'text-indigo-700 font-medium' : 'text-gray-500'}`}>

                          {domain.need}

                        </p>

                      </div>

                      <div className="text-right flex-shrink-0">

                        <span className={`text-2xl font-bold ${index === 0 ? 'text-indigo-700' : 'text-gray-600'}`}>

                          {domain.averageScore}

                        </span>

                        <span className="text-xs text-gray-400 block">средний балл</span>

                      </div>

                    </div>

                  </div>

                ))}

              </div>

              <p className="text-sm text-gray-500 mt-4 italic">

                * Домен с наивысшим баллом указывает на главную внутреннюю потребность, требующую проработки в коучинге.

              </p>

            </div>


            {/* Блок Метапрограмм */}

            <div className="bg-white p-8 rounded-2xl shadow-sm border border-gray-100">

              <h3 className="text-xl font-bold text-gray-800 mb-6 border-b pb-4">Ваши ведущие Метапрограммы</h3>

              

              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">

                {calculateResults.metaprograms.filter(mp => mp.total >= 5).map((mp) => (

                  <div key={mp.id} className={`p-4 rounded-xl border ${mp.total >= 8 ? 'border-red-200 bg-red-50' : 'border-yellow-200 bg-yellow-50'}`}>

                    <div className="flex justify-between items-start mb-3">

                      <h4 className="font-bold text-gray-900 pr-2">{mp.id}. {mp.name}</h4>

                      <span className="bg-white px-2 py-1 rounded text-lg font-bold shadow-sm">{mp.total} <span className="text-xs text-gray-400 font-normal">/10</span></span>

                    </div>

                    <div className="flex items-center text-sm">

                      <span className={`px-2 py-0.5 rounded-full font-medium text-xs ${mp.bg} ${mp.color}`}>

                        {mp.severity}

                      </span>

                      <span className="text-gray-500 ml-2 text-xs truncate" title={mp.domainName}>

                        {mp.domainName.split(':')[1]}

                      </span>

                    </div>

                  </div>

                ))}

              </div>

              

              {calculateResults.metaprograms.filter(mp => mp.total >= 5).length === 0 && (

                <div className="text-center p-8 bg-green-50 rounded-xl border border-green-200 text-green-800">

                  <p className="font-semibold text-lg">У вас нет ярко выраженных дезадаптивных метапрограмм.</p>

                  <p className="mt-2 text-sm">Это означает высокую психологическую гибкость и устойчивость.</p>

                </div>

              )}

            </div>


            <div className="text-center pt-8">

              <button 

                onClick={resetTest}

                className="px-6 py-3 bg-white border-2 border-indigo-600 text-indigo-600 hover:bg-indigo-50 font-bold rounded-xl transition-all"

              >

                Пройти тест заново

              </button>

            </div>

          </div>

        )}

      </main>

      <style dangerouslySetInnerHTML={{__html: `

        @keyframes fadeIn {

          from { opacity: 0; transform: translateY(10px); }

          to { opacity: 1; transform: translateY(0); }

        }

        .animate-fade-in {

          animation: fadeIn 0.4s ease-out forwards;

        }

      `}} />

    </div>

  );
